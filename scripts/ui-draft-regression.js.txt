async page => {
  const close = page.getByRole('button', {name:'关闭任务详情',exact:true});
  if (await close.count()) await close.click();
  await page.getByRole('button', {name:'新建本地任务',exact:true}).click();
  await page.getByRole('textbox', {name:'任务标题',exact:true}).fill('草稿保留回归');
  await page.getByRole('button', {name:'创建任务',exact:true}).click();
  await page.getByRole('heading', {name:'草稿保留回归',level:2,exact:true}).waitFor();
  await page.getByRole('textbox', {name:'任务标题',exact:true}).fill('UNSAVED-FIELD-MUST-SURVIVE');
  await page.getByRole('button', {name:'转为 LinkinNotes 个人待办',exact:true}).click();
  await page.getByRole('button', {name:'已关联，不会重复创建',exact:true}).waitFor();
  const value = await page.getByRole('textbox', {name:'任务标题',exact:true}).inputValue();
  if (value !== 'UNSAVED-FIELD-MUST-SURVIVE') throw new Error('Link action discarded draft');
  await page.getByRole('button', {name:'保存修改',exact:true}).click();
  await page.getByRole('heading', {name:'UNSAVED-FIELD-MUST-SURVIVE',level:2,exact:true}).waitFor();
  await page.getByRole('button', {name:'关闭任务详情',exact:true}).click();
  return {result:'unsaved field survives task linking and can be saved afterwards'};
}
