async page => {
  await page.setViewportSize({width:1440,height:1000});
  const api=async(method,params={})=>{const r=await page.request.post(new URL('/rpc',page.url()).href,{data:{method,params}});const v=await r.json();if(v.error)throw new Error(v.error);return v.result;};
  const {project}=await api('demo');
  const settings=(await api('snapshot')).settings;
  await api('settings_save',{...settings,people:[{id:'开发',name:'开发',department:'开发'},{id:'UI',name:'UI',department:'UI'}]});
  await page.evaluate(id=>localStorage.setItem('larkpm-project',id),project);await page.reload();
  const choose=async(label,value)=>{await page.getByRole('combobox',{name:label,exact:true}).click();await page.getByRole('listbox',{name:label+'选项',exact:true}).getByRole('option',{name:value,exact:true}).click();};
  const heights=await page.locator('.workflow-controls .search,.workflow-controls .filterbar>.list-select,.workflow-controls .button,.workflow-shortcuts').evaluateAll(els=>els.map(el=>({height:el.getBoundingClientRect().height,radius:getComputedStyle(el).borderRadius,font:getComputedStyle(el).fontSize})));
  if(heights.some(el=>el.height!==34||el.radius!=='6px'||el.font!=='11px'))throw new Error('Toolbar control styles inconsistent: '+JSON.stringify(heights));
  await choose('按处理人筛选','开发');
  if(await page.locator('.flow-node').count()!==2)throw new Error('Owner custom list failed');
  await page.getByRole('combobox',{name:'按处理人筛选',exact:true}).click();
  const list=page.getByRole('listbox',{name:'按处理人筛选选项',exact:true});
  const selected=list.getByRole('option',{name:'开发',exact:true});
  if(await selected.getAttribute('aria-selected')!=='true'||!await selected.locator('svg').count())throw new Error('Selected row check missing');
  if(await selected.evaluate(el=>getComputedStyle(el).backgroundColor)!=='rgb(237, 237, 237)')throw new Error('Selected row not gray');
  await page.screenshot({path:'output/playwright/select-list-reference.png'});
  await page.keyboard.press('Escape');
  await page.getByRole('combobox',{name:'按处理人筛选',exact:true}).focus();
  await page.keyboard.press('ArrowDown');await page.keyboard.press('Home');await page.keyboard.press('Enter');
  if(await page.getByRole('combobox',{name:'按处理人筛选',exact:true}).innerText()!=='全部处理人')throw new Error('List keyboard selection failed');
  await choose('按部门筛选','UI');
  if(await page.locator('.flow-node').count()!==1)throw new Error('Department list filtering failed');
  await choose('按部门筛选','全部部门');
  await choose('按版本筛选','v1.0');
  await choose('按版本筛选','全部版本');
  await page.getByRole('button',{name:'查看UI绘制',exact:true}).click();
  const dialog=page.getByRole('dialog',{name:'任务详情',exact:true});
  await dialog.waitFor();
  if(await dialog.getByRole('textbox').count())throw new Error('Duplicate edit form visible by default');
  const box=await dialog.boundingBox();
  if(box.width>602||box.height>650||Math.abs(box.x+box.width/2-720)>3)throw new Error('Dialog not compact/centered: '+JSON.stringify(box));
  await page.screenshot({path:'output/playwright/task-inline-compact.png'});
  await dialog.getByRole('button',{name:'编辑任务标题',exact:true}).dblclick();
  await dialog.getByRole('textbox',{name:'任务标题',exact:true}).fill('UI原位编辑');
  await dialog.getByRole('button',{name:'编辑所属模块',exact:true}).dblclick();
  await dialog.getByRole('textbox',{name:'所属模块',exact:true}).fill('紧凑系统');
  await dialog.getByRole('button',{name:'编辑开始时间',exact:true}).dblclick();
  await dialog.getByLabel('开始时间',{exact:true}).fill('2026-11-06');
  await dialog.getByRole('button',{name:'编辑截止时间',exact:true}).dblclick();
  await dialog.getByLabel('截止时间',{exact:true}).fill('2026-11-09');
  await dialog.getByRole('button',{name:'编辑版本',exact:true}).dblclick();
  await dialog.getByLabel('版本',{exact:true}).fill('v9');
  await dialog.getByRole('button',{name:'编辑处理人',exact:true}).dblclick();
  await choose('添加已有处理人','开发');
  await dialog.getByRole('button',{name:'编辑需求部门',exact:true}).dblclick();
  await dialog.getByRole('checkbox',{name:'使用人员默认部门',exact:true}).uncheck();
  await choose('手动指定部门','动效');
  await dialog.getByRole('button',{name:'编辑工作状态',exact:true}).dblclick();
  await choose('工作状态','暂停');
  await dialog.getByRole('button',{name:'保存修改',exact:true}).click();
  await dialog.getByText('任务已保存于本地',{exact:true}).waitFor();
  const task=(await api('snapshot',{project})).tasks.find(t=>t.title==='UI原位编辑');
  if(!task||task.module!=='紧凑系统'||task.version!=='v9'||task.start!=='2026-11-06'||task.end!=='2026-11-09'||task.state!=='paused'||!task.owners.some(p=>p.id==='开发')||!task.departments.includes('动效'))throw new Error('Inline edits not persisted');
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  await page.reload();await page.getByRole('button',{name:'查看UI原位编辑',exact:true}).click();
  await dialog.getByRole('button',{name:'编辑任务标题',exact:true}).focus();await page.keyboard.press('Enter');
  await dialog.getByRole('textbox',{name:'任务标题',exact:true}).waitFor();
  await page.keyboard.press('Escape');
  if(!await dialog.isVisible()||await dialog.getByRole('textbox',{name:'任务标题',exact:true}).count())throw new Error('Inline Escape did not retain modal');
  await dialog.getByText('前置依赖 · 1 项',{exact:true}).click();
  await dialog.getByText('编辑前置任务',{exact:true}).click();
  await dialog.getByRole('textbox',{name:'搜索前置任务',exact:true}).fill('需求文档');
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  await page.setViewportSize({width:1000,height:720});
  await page.getByRole('button',{name:'查看UI原位编辑',exact:true}).click();
  const small=await dialog.boundingBox();if(small.x<0||small.y<0||small.y+small.height>720)throw new Error('Small dialog outside viewport');
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  return 'Uniform toolbar, reference list/checkmark/filter behavior, compact modal, original-position edits for eight fields, saved values, keyboard editing, dependencies and small viewport passed';
}
