async page => {
  await page.setViewportSize({width:1440,height:1000});
  const api=async(method,params={})=>{const response=await page.request.post(new URL('/rpc',page.url()).href,{data:{method,params}});const value=await response.json();if(value.error)throw new Error(value.error);return value.result;};
  const {project}=await api('demo');
  const settings=(await api('snapshot')).settings;
  await api('settings_save',{...settings,profile:{name:'小美',owner_id:'UI'},people:[{id:'UI',name:'小美',department:'UI'},{id:'开发',name:'小开',department:'开发'},{id:'departed',name:'离职人员',department:'UI',employment_status:'resigned'}]});
  let ui=(await api('snapshot',{project})).tasks.find(task=>task.title==='UI绘制');
  await api('edit',{id:ui.id,revision:ui.revision,patch:{department_override:['策划']}});
  const added=await api('create',{project,title:'批量新增',owners:[{id:'UI',name:'小美'}],module:'额外需求'});
  const linked=await api('create',{project,title:'已关联记录',owners:[{id:'UI',name:'小美'}],module:'额外需求'});
  await api('link',{id:linked.id,revision:linked.revision});
  await page.evaluate(id=>localStorage.setItem('larkpm-project',id),project);await page.reload();
  await page.getByRole('button',{name:'查看UI绘制',exact:true}).click();
  const dialog=page.getByRole('dialog',{name:'任务详情',exact:true});await dialog.waitFor();
  for(const text of ['飞书需求','本地任务','双击内容修改','原型图与流程图已可用','使用人员 ID 添加','LinkinNotes 个人待办'])if(await dialog.getByText(text,{exact:true}).count())throw new Error('Removed UI still present: '+text);
  if(await dialog.locator('details').first().getAttribute('open')===null)throw new Error('Upstream not expanded');
  await dialog.getByRole('button',{name:'编辑任务标题',exact:true}).click();
  await dialog.getByRole('textbox',{name:'任务标题',exact:true}).fill('UI单击修改');
  const owners=dialog.getByRole('combobox',{name:'处理人',exact:true});await owners.click();
  const list=page.getByRole('listbox',{name:'处理人选项',exact:true});
  if(await list.getByRole('option',{name:/离职人员/}).count())throw new Error('Departed person shown');
  await list.getByRole('option',{name:'小开 · 开发',exact:true}).click();
  if(!await list.isVisible())throw new Error('Multi-person menu closed after first choice');
  await page.keyboard.press('Escape');
  const departments=dialog.getByRole('status',{name:'处理人部门',exact:true});
  await departments.getByText('UI',{exact:true}).waitFor();await departments.getByText('开发',{exact:true}).waitFor();
  const ownerBox=await owners.boundingBox(),deptBox=await departments.boundingBox();
  if(Math.abs(ownerBox.y-deptBox.y)>4)throw new Error('Owner/department not in same row');
  if(await dialog.getByRole('button',{name:'编辑需求部门',exact:true}).count())throw new Error('Department remains independently editable');
  await dialog.getByRole('button',{name:'保存修改',exact:true}).click();await dialog.getByText('任务已保存于本地',{exact:true}).waitFor();
  ui=(await api('snapshot',{project})).tasks.find(task=>task.id===ui.id);
  if(ui.title!=='UI单击修改'||ui.department_override!==null||ui.departments.length!==2||ui.owners.length!==2)throw new Error('Owner/automatic department not saved');
  await page.screenshot({path:'output/playwright/task-single-click-people.png'});
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  await page.getByRole('button',{name:'个人工作台',exact:true}).click();
  if(await page.getByRole('checkbox',{name:'选择任务：功能开发',exact:true}).count())throw new Error('Personal identity filtering lost');
  if(!await page.getByRole('checkbox',{name:'选择任务：已关联记录',exact:true}).isDisabled())throw new Error('Already linked task selectable');
  await page.getByRole('checkbox',{name:'选择任务：UI单击修改',exact:true}).check();
  await page.getByRole('checkbox',{name:'选择任务：批量新增',exact:true}).check();
  await page.getByText('已选 2 项',{exact:true}).waitFor();
  const before=(await api('snapshot',{project})).tasks.length;
  await page.getByRole('button',{name:'转为 LinkinNotes',exact:true}).click();
  await page.getByText('所选任务已转为 LinkinNotes 个人待办',{exact:true}).waitFor();
  const snapshot=await api('snapshot',{project});
  if(snapshot.tasks.length!==before||![ui.id,added.id,linked.id].every(id=>snapshot.tasks.find(task=>task.id===id).linked))throw new Error('Batch link did not preserve original records');
  if(!await page.getByRole('button',{name:'转为 LinkinNotes',exact:true}).isDisabled())throw new Error('Converted selection not cleared');
  await page.reload();await page.getByRole('button',{name:'个人工作台',exact:true}).click();
  if(!await page.getByRole('checkbox',{name:'选择任务：UI单击修改',exact:true}).isDisabled())throw new Error('Link not persisted');
  await page.screenshot({path:'output/playwright/personal-batch-link.png'});
  return 'Single-click editor, removed header/prototype/IDs/conversion, expanded dependencies, multi-person list and same-row automatic departments, persisted batch LinkinNotes and personal identity passed';
}
