async page => {
  await page.setViewportSize({width:1440,height:1000});
  const api=async(method,params={})=>{const r=await page.request.post(new URL('/rpc',page.url()).href,{data:{method,params}});const v=await r.json();if(v.error)throw new Error(v.error);return v.result;};
  const choose=async(label,value)=>{await page.getByRole('combobox',{name:label,exact:true}).click();await page.getByRole('listbox',{name:label+'选项',exact:true}).getByRole('option',{name:value,exact:true}).click();};
  const {project}=await api('demo');
  const a=await api('create',{project,title:'完成记录A',module:'已结系统',state:'done'});
  const b=await api('create',{project,title:'关闭记录B',module:'已结系统',state:'closed'});
  const done=await api('create',{project,title:'同系统已完成',module:'继续系统',state:'done',owners:[{id:'alice',name:'甲'}]});
  const pending=await api('create',{project,title:'同系统待处理',module:'继续系统',owners:[{id:'bob',name:'乙'}]});
  await api('workflow_column_save',{project,name:'空需求列',priority:'high'});
  const other=await api('project_save',{name:'归档隔离项目',url:'https://example.feishu.cn/base/baseArchived?table=tblArchived'});
  await api('create',{project:other.id,title:'别的项目任务',module:'别的项目已结',state:'done'});
  await page.evaluate(id=>localStorage.setItem('larkpm-project',id),project);await page.reload();
  const col=name=>page.getByRole('region',{name:name+'需求流程',exact:true});
  const archiveButton=page.getByRole('button',{name:/^已结需求/});
  await col('继续系统').waitFor();
  if(await col('已结系统').count() || await page.locator('.state-done,.state-closed').count())throw new Error('Finished requirements/tasks still on board');
  await col('空需求列').waitFor();
  if(await archiveButton.innerText()!=='已结需求 · 1')throw new Error('Empty column archived or count incorrect');
  await page.keyboard.down('Alt');
  if(await col('已结系统').count()||await page.locator('.state-done,.state-closed').count())throw new Error('Finished requirements/tasks visible in arrangement');
  await page.keyboard.up('Alt');
  await choose('按处理人筛选','甲');
  await archiveButton.click();
  let modal=page.getByRole('dialog',{name:/已结需求/});
  await modal.waitFor();
  if(await modal.getByText('继续系统',{exact:true}).count() || await modal.getByText('别的项目已结',{exact:true}).count())throw new Error('Filtering or project scope incorrectly classified archive');
  if(await modal.getByRole('button',{name:'查看历史任务完成记录A',exact:true}).isVisible())throw new Error('Archive expanded by default');
  await modal.locator('summary').filter({hasText:'已结系统'}).click();
  await modal.getByRole('button',{name:'查看历史任务完成记录A',exact:true}).waitFor();
  await page.screenshot({path:'output/playwright/archived-requirements.png'});
  await modal.getByRole('textbox',{name:'搜索已结需求',exact:true}).fill('不存在');
  await modal.getByText('没有匹配的已结需求',{exact:true}).waitFor();
  await modal.getByRole('textbox',{name:'搜索已结需求',exact:true}).fill('');
  await modal.getByRole('button',{name:'追加任务到已结系统',exact:true}).click();
  const create=page.getByRole('dialog',{name:'新建本地任务',exact:true});
  if(await create.getByRole('combobox',{name:'所属模块',exact:true}).inputValue()!=='已结系统')throw new Error('Append did not preserve system');
  await create.getByRole('textbox',{name:'任务标题',exact:true}).fill('新增需求');
  await create.getByRole('button',{name:'创建任务',exact:true}).click();
  await page.getByRole('dialog',{name:'任务详情',exact:true}).waitFor();
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  await choose('按处理人筛选','全部处理人');
  await col('已结系统').waitFor();
  if(await col('已结系统').locator('.flow-node').count()!==1)throw new Error('Old finished tasks revealed by append');
  if(await archiveButton.innerText()!=='已结需求')throw new Error('Reopened system remained archived');
  await page.getByRole('button',{name:'切换状态：新增需求',exact:true}).click();
  await page.getByRole('dialog',{name:'快捷流转状态',exact:true}).getByRole('button',{name:'交付就绪',exact:true}).click();
  await col('已结系统').waitFor({state:'hidden'});
  await page.reload();await archiveButton.click();
  modal=page.getByRole('dialog',{name:/已结需求/});
  await modal.locator('summary').filter({hasText:'已结系统'}).click();
  await modal.getByRole('button',{name:'查看历史任务完成记录A',exact:true}).click();
  await page.getByRole('button',{name:'编辑工作状态',exact:true}).dblclick();
  await choose('工作状态','未开始');
  await page.getByRole('button',{name:'保存修改',exact:true}).click();
  await page.getByText('任务已保存于本地',{exact:true}).waitFor();
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  await col('已结系统').waitFor();
  const snapshot=await api('snapshot',{project});
  if(![a.id,b.id,done.id,pending.id].every(id=>snapshot.tasks.some(t=>t.id===id)))throw new Error('History lost');
  await archiveButton.click();await page.keyboard.press('Escape');
  if(await page.getByRole('dialog',{name:/已结需求/}).count())throw new Error('Archive Escape close failed');
  const search=await page.getByRole('textbox',{name:'搜索任务或模块',exact:true}).boundingBox();
  const button=await page.getByRole('button',{name:'新建需求列',exact:true}).boundingBox();
  if(Math.abs(search.y-button.y)>4)throw new Error('Added archive entry split controls into rows');
  await page.screenshot({path:'output/playwright/requirements-after-reopen.png'});
  return 'Completed task hiding, automatic requirement archive/reopen, modal history/search/append, full-data classification, project isolation, empty columns and retained history passed';
}
