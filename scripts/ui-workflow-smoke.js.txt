async page => {
  await page.setViewportSize({width:1440,height:1000});
  await page.clock.install({time:new Date('2026-12-31T12:00:00+08:00')});
  const api=async(method,params={})=>{const r=await page.request.post(new URL('/rpc',page.url()).href,{data:{method,params}});const v=await r.json();if(v.error)throw new Error(v.error);return v.result;};
  const {project}=await api('demo');
  const settings=(await api('snapshot')).settings;
  await api('settings_save',{...settings,people:[{id:'开发',name:'开发',department:'开发'},{id:'UI',name:'UI',department:'UI'},{id:'策划',name:'策划',department:'策划'},{id:'动效',name:'动效',department:'动效'},{id:'切摆',name:'切摆',department:'UI'}]});
  for(const task of (await api('snapshot')).tasks.filter(t=>t.project===project && t.state==='active')) {
    await api('edit',{id:task.id,revision:task.revision,patch:{end:task.title==='功能开发'?'2027-01-02':task.title==='UI绘制'?'2026-12-30':'2026-12-31'}});
  }
  await page.evaluate(id=>localStorage.setItem('larkpm-project',id),project);
  await page.reload();
  await page.locator('.workflow-board').waitFor({timeout:3000});
  if(await page.locator('.flow-node.is-active').count()!==3)throw new Error('Not all parallel active tasks expanded');
  if(await page.locator('.flow-node:not(.is-active) .flow-node-detail').count())throw new Error('Inactive nodes show expanded detail');
  const parallel=page.locator('.flow-stage').filter({hasText:'功能开发'});
  if(!await parallel.getByText('并行 · 3 个节点',{exact:true}).count())throw new Error('Parallel stage missing');
  if(!await parallel.getByText('UI绘制',{exact:true}).count())throw new Error('Parallel work incorrectly serialized');
  const stages=await page.locator('.flow-stage').count();
  if(stages!==3)throw new Error('Dependency levels incorrect: '+stages);
  const active=page.locator('.flow-node').filter({hasText:'功能开发'});
  if(!await active.locator('.people').count() || !await active.locator('.department-tag').count() || !await active.locator('.flow-deadline').count())throw new Error('Essential active metadata missing');
  const card=active.locator('.flow-node-card');
  if((await card.boundingBox()).height>90)throw new Error('Active card too tall');
  const style=await card.evaluate(el=>({background:getComputedStyle(el).backgroundColor,color:getComputedStyle(el).color}));
  if(style.background!=='rgb(255, 255, 255)')throw new Error('Active card still has a colored background');
  if(await page.locator('.flow-field-label,.flow-node-footer,.workflow-progress,.workflow-eyebrow').count())throw new Error('Redundant fields remain');
  if(await active.locator('.flow-node-heading .department-tag').count()!==1)throw new Error('Department not next to title');
  if(await active.locator('.flow-node-detail .department-tag,.flow-blocked').count())throw new Error('Redundant department or dependency detail');
  if(!await active.locator('.flow-deadline').getByText('2027/01/02（2d）',{exact:true}).count())throw new Error('Year boundary countdown wrong');
  const overdue=page.locator('.flow-node').filter({hasText:'UI绘制'}).locator('.flow-deadline');
  if(await overdue.innerText()!=='2026/12/30（已延期 1d）' || !await overdue.evaluate(el=>getComputedStyle(el).color==='rgb(207, 56, 56)'))throw new Error('Overdue label/color wrong');
  if(!await card.evaluate(el=>getComputedStyle(el).borderTopWidth==='2px'))throw new Error('Department border not emphasized');
  const compact=page.locator('.flow-node.is-compact .flow-node-card').first();
  if((await compact.boundingBox()).height>50)throw new Error('Compact node too tall');
  await page.locator('.main-content').evaluate(el=>el.scrollTop=140);
  await page.screenshot({path:'output/playwright/workflow-overview.png'});
  await page.getByRole('button',{name:'查看预制件动效制作',exact:true}).click();
  await page.getByRole('dialog',{name:'任务详情',exact:true}).waitFor();
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  await page.getByRole('textbox',{name:'搜索任务或模块',exact:true}).fill('开发接入动效');
  if(await page.locator('.flow-node').count()!==1)throw new Error('Filtering lost');
  await page.getByRole('button',{name:'查看开发接入动效',exact:true}).click();
  await page.getByRole('combobox',{name:'工作状态',exact:true}).selectOption('active');
  await page.getByRole('button',{name:'保存修改',exact:true}).click();
  await page.getByText('上游尚未完成，请先完成上游或调整依赖',{exact:true}).waitFor();
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  for(const task of (await api('snapshot',{project})).tasks.filter(t=>t.title!=='开发接入动效'))await api('edit',{id:task.id,revision:task.revision,patch:{state:'done'}});
  await page.reload();
  await page.getByRole('textbox',{name:'搜索任务或模块',exact:true}).fill('开发接入动效');
  await page.getByRole('button',{name:'查看开发接入动效',exact:true}).click();
  await page.getByRole('combobox',{name:'工作状态',exact:true}).selectOption('active');
  await page.getByRole('button',{name:'保存修改',exact:true}).click();
  await page.getByText('任务已保存于本地',{exact:true}).waitFor();
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  if(await page.locator('.flow-node.is-active .flow-node-detail').count()!==1)throw new Error('State change did not expand node');
  await page.getByRole('textbox',{name:'搜索任务或模块',exact:true}).fill('');
  for(const module of ['背包系统','角色系统','商城系统'])await api('create',{project,title:module+'需求',module,version:'v1.2'});
  await page.reload();
  await page.locator('.workflow-module').nth(3).waitFor();
  const layout=await page.locator('.workflow-board').evaluate(el=>({width:el.clientWidth,scroll:el.scrollWidth,columns:[...el.children].map(c=>c.getBoundingClientRect().width)}));
  if(layout.scroll<=layout.width || layout.columns.some(width=>Math.abs(width-330)>1))throw new Error('Columns not fixed at 330px with horizontal overflow');
  await page.setViewportSize({width:1920,height:1000});
  const wide=await page.locator('.workflow-board').evaluate(el=>({width:el.clientWidth,columns:[...el.children].map(c=>c.getBoundingClientRect().width)}));
  if(wide.columns.some(width=>Math.abs(width-330)>1) || Math.floor((wide.width+12)/342)<=Math.floor((layout.width+12)/342))throw new Error('Wider window did not reveal more fixed-width columns');
  await page.setViewportSize({width:1440,height:1000});
  if(await page.locator('.workflow-versions').getByText('v1.2',{exact:true}).count()!==3)throw new Error('Module versions missing');
  await page.locator('.main-content').evaluate(el=>el.scrollTop=140);
  await page.screenshot({path:'output/playwright/workflow-overview.png'});
  await page.locator('.workflow-board').evaluate(el=>el.scrollLeft=el.scrollWidth);
  await page.getByRole('button',{name:'查看商城系统需求',exact:true}).click();
  await page.getByRole('dialog',{name:'任务详情',exact:true}).waitFor();
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  return 'department/title layout, deadline countdown/overdue, three-column horizontal scrolling, module versions and workflow interactions passed';
}
