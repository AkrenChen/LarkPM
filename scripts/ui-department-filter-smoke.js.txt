async page => {
  await page.setViewportSize({width:1320,height:900});
  const api=async(method,params={})=>{const r=await page.request.post(new URL('/rpc',page.url()).href,{data:{method,params}});const v=await r.json();if(v.error)throw new Error(v.error);return v.result;};
  const p=await api('project_save',{name:'部门筛选验收',url:'https://example.feishu.cn/base/baseF?table=tblF'});
  const other=await api('project_save',{name:'其他项目',url:'https://example.feishu.cn/base/baseOther?table=tblOther'});
  for(const [title,department,id,name,version] of [['功能开发','开发','a','甲','v1'],['接入动效','开发','a','甲','v2'],['UI绘制','UI','b','乙','v1'],['界面检查','UI','c','丙','v1']])await api('create',{project:p.id,title,department_override:[department],owners:[{id,name}],module:'活动系统',version,start:'2026-11-01',end:'2026-11-05',state:'active'});
  const settings=(await api('snapshot')).settings;
  await api('settings_save',{...settings,profile:{name:'甲',owner_id:'a'}});
  await page.evaluate(id=>localStorage.setItem('larkpm-project',id),p.id);
  await page.reload();
  const choose=async name=>{await page.getByRole('combobox',{name:'按部门筛选',exact:true}).click();await page.getByRole('listbox',{name:'按部门筛选选项',exact:true}).getByRole('option',{name,exact:true}).click();};
  const owner=page.getByRole('combobox',{name:'按处理人筛选',exact:true});
  await choose('开发');
  if(await page.locator('.flow-node').count()!==2 || JSON.stringify(await owner.locator('option').allTextContents())!==JSON.stringify(['全部处理人','甲']))throw new Error('Department task/person filtering failed');
  await owner.selectOption('a');
  await choose('UI');
  if(await owner.inputValue()!=='' || await page.locator('.flow-node').count()!==2 || JSON.stringify(await owner.locator('option').allTextContents())!==JSON.stringify(['全部处理人','乙','丙']))throw new Error('Switching department did not reset incompatible owner');
  await owner.selectOption('b');
  if(await page.locator('.flow-node').count()!==1 || !await page.getByRole('button',{name:'查看UI绘制',exact:true}).count())throw new Error('Combined department/person filtering failed');
  await page.getByRole('button',{name:'版本甘特',exact:true}).click();
  if(await owner.inputValue()!=='b' || await page.locator('.gantt-bar').count()!==1)throw new Error('Gantt did not keep filters');
  await page.getByRole('button',{name:'个人工作台',exact:true}).click();
  if(await owner.inputValue()!=='a' || await page.locator('.task-row').count()!==2)throw new Error('Personal default scope changed');
  await choose('UI');
  await owner.selectOption('c');
  if(await page.locator('.task-row').count()!==1)throw new Error('Personal department/person filtering failed');
  await page.getByRole('button',{name:'我的任务',exact:true}).click();
  if(await owner.inputValue()!=='a' || await page.locator('.task-row').count()!==2)throw new Error('My tasks did not restore own scope');
  await page.getByRole('button',{name:'需求总览',exact:true}).click();
  await choose('全部部门');
  await owner.selectOption('');
  await page.getByRole('combobox',{name:'按版本筛选',exact:true}).selectOption('v2');
  await choose('UI');
  if(await page.locator('.flow-node').count())throw new Error('Version/department intersection failed');
  await page.getByRole('combobox',{name:'按版本筛选',exact:true}).selectOption('');
  const padding=await page.locator('.main-content').evaluate(el=>{const s=getComputedStyle(el);return [s.paddingTop,s.paddingLeft,s.paddingRight];});
  if(JSON.stringify(padding)!==JSON.stringify(['14px','18px','18px']))throw new Error('Workspace margins not compact');
  await page.screenshot({path:'output/playwright/department-filter.png'});
  await page.setViewportSize({width:1920,height:1000});
  if(await page.locator('.main-content').evaluate(el=>getComputedStyle(el).paddingLeft)!=='18px')throw new Error('Wide layout padding overrides compact setting');
  await page.getByRole('combobox',{name:'当前项目',exact:true}).selectOption(other.id);
  if(await page.getByRole('combobox',{name:'按部门筛选',exact:true}).innerText()!=='全部部门' || await owner.inputValue()!=='')throw new Error('Filters leaked between projects');
  return 'department/person cascade, invalid-owner reset, three views, personal identity, version intersection, project isolation and compact padding passed';
}
