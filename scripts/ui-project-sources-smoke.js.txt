async page => {
  await page.setViewportSize({width:1440,height:1000});
  const api=async(method,params={})=>{const r=await page.request.post(new URL('/rpc',page.url()).href,{data:{method,params}});const v=await r.json();if(v.error)throw new Error(v.error);return v.result;};
  const p=await api('project_save',{name:'项目关联验收',url:'https://example.feishu.cn/base/baseP?table=tblP',sources:{tests:{url:'https://example.feishu.cn/base/baseT?table=tblT',project_value:'SolitairePuzzle'},versions:{url:'https://example.feishu.cn/base/baseV?table=tblV'}}});
  for(const [state,title] of [['paused','方案评审'],['closed','旧版方案'],['done','需求文档'],['active','功能开发'],['todo','UI绘制']])await api('create',{project:p.id,state,title,module:'活动系统',start:'2026-11-01',end:'2026-11-04',owners:[{id:'me',name:'测试人员'}]});
  const settings=(await api('snapshot')).settings;
  await api('settings_save',{...settings,profile:{name:'测试人员',owner_id:'me'}});
  await api('project_sources_sync',{project:p.id});
  await page.evaluate(id=>localStorage.setItem('larkpm-project',id),p.id);
  await page.reload();
  if(await page.getByRole('button',{name:'项目详情',exact:true}).count())throw new Error('Standalone project details remains');
  if(await page.locator('.topbar').getByRole('button',{name:'读取飞书',exact:true}).count())throw new Error('Read button still in topbar');
  const actions=page.locator('.page-heading-actions');
  if(!await actions.getByRole('button',{name:'读取飞书',exact:true}).count() || !await actions.getByRole('button',{name:'新建本地任务',exact:true}).count())throw new Error('Read/create actions not grouped');
  if(await page.locator('.metrics').count())throw new Error('Old statistic cards remain');
  const widths=await page.locator('.status-progress-segment').evaluateAll(els=>els.map(el=>el.style.width));
  if(JSON.stringify(widths)!==JSON.stringify(['40%','20%','20%','20%']))throw new Error('Progress proportions wrong: '+widths);
  if(!await page.getByText('暂停 1 项／已关闭 1 项',{exact:true}).count())throw new Error('Paused/closed count missing');
  await page.getByText('v1.0.9',{exact:true}).waitFor();
  if(await page.getByText('v1.0.10',{exact:true}).count())throw new Error('Unapproved version presented as online');
  if(!await page.getByText('暂无已过审版本',{exact:true}).count())throw new Error('Missing iOS online fallback');
  await page.getByRole('button',{name:'需求表 ↗',exact:true}).click();
  if((await page.evaluate(()=>window.__openedProjectTable)).kind!=='requirements')throw new Error('Wrong source open action');
  await page.getByRole('button',{name:'版本表 ↗',exact:true}).click();
  if((await page.evaluate(()=>window.__openedProjectTable)).kind!=='versions')throw new Error('Wrong version open action');
  await page.screenshot({path:'output/playwright/project-summary.png'});
  await page.getByRole('button',{name:'个人工作台',exact:true}).click();
  await page.getByRole('region',{name:'项目线上版本与源表',exact:true}).waitFor();
  await page.getByRole('region',{name:'任务进度统计',exact:true}).waitFor();
  await page.getByRole('button',{name:'版本甘特',exact:true}).click();
  await page.getByRole('region',{name:'项目线上版本与源表',exact:true}).waitFor();
  await page.getByRole('region',{name:'任务进度统计',exact:true}).waitFor();
  await page.locator('.gantt-test-bar').nth(1).waitFor();
  if(await page.locator('.gantt-test-bar').count()!==2 || await page.getByText('其他项目测试',{exact:true}).count())throw new Error('Testing project isolation failed');
  if((await api('snapshot')).tasks.filter(t=>t.project===p.id).length!==5)throw new Error('Tests became tasks');
  await page.getByRole('button',{name:'刷新测试排期',exact:true}).click();
  await page.screenshot({path:'output/playwright/project-test-gantt.png'});
  await page.getByRole('button',{name:'设置',exact:true}).click();
  await page.getByRole('button',{name:'项目与连接',exact:true}).click();
  await page.getByLabel('测试表中的项目名',{exact:true}).fill('Solitaire');
  await page.getByRole('button',{name:'保存项目连接',exact:true}).click();
  await page.getByText('项目连接已保存',{exact:true}).waitFor();
  await page.getByRole('button',{name:'版本甘特',exact:true}).click();
  if(await page.locator('.gantt-test-bar').count())throw new Error('Previous project test cache survived mapping change');
  await page.getByRole('button',{name:'刷新测试排期',exact:true}).click();
  await page.locator('.gantt-test-bar').waitFor();
  if(await page.locator('.gantt-test-bar').count()!==1)throw new Error('Updated project mapping ignored');
  return 'per-platform online versions, links, testing-only Gantt without tasks, exact project match, configurable mapping and cache invalidation passed';
}
