async page => {
  await page.setViewportSize({width:1440,height:1000});
  const api=async(method,params={})=>{const r=await page.request.post(new URL('/rpc',page.url()).href,{data:{method,params}});const v=await r.json();if(v.error)throw new Error(v.error);return v.result;};
  const choose=async(label,value)=>{await page.getByRole('combobox',{name:label,exact:true}).click();await page.getByRole('listbox',{name:label+'选项',exact:true}).getByRole('option',{name:value,exact:true}).click();};
  const {project}=await api('demo');
  const a=await api('create',{project,title:'待归类A'}),b=await api('create',{project,title:'待归类B'}),c=await api('create',{project,title:'待归类C'});
  await page.evaluate(id=>localStorage.setItem('larkpm-project',id),project);await page.reload();
  const col=name=>page.getByRole('region',{name:name+'需求流程',exact:true});
  const node=id=>page.locator('[data-task-id="'+id+'"]');
  const waitFor=async(fn,label)=>{for(let i=0;i<40;i++){if(await fn())return;await page.waitForTimeout(100);}throw new Error(label);};
  await page.getByRole('button',{name:'新建需求列',exact:true}).click();
  await page.getByRole('textbox',{name:'需求列名称',exact:true}).fill('交互系统');
  await choose('优先级','高');
  await page.getByRole('button',{name:'保存需求列',exact:true}).click();
  await col('交互系统').waitFor();
  if(await page.locator('.workflow-module h2').first().innerText()!=='交互系统')throw new Error('Priority order');
  await page.reload();await col('交互系统').waitFor();
  await page.getByRole('button',{name:'查看待归类A',exact:true}).click({modifiers:['Control']});
  await page.getByRole('button',{name:'查看待归类B',exact:true}).click({modifiers:['Control']});
  await page.keyboard.down('Alt');
  await node(a.id).dragTo(col('交互系统').locator('[data-drop-gap="0"]'));
  await page.keyboard.up('Alt');
  await waitFor(async()=>{const s=await api('snapshot',{project});return s.tasks.find(t=>t.id===a.id).module==='交互系统';},'Multi drag did not save');
  await waitFor(async()=>await col('交互系统').locator('.flow-node').count()===2,'Moved cards missing');
  let snapshot=await api('snapshot',{project});
  if(snapshot.tasks.find(t=>t.id===b.id).workflow_stage!==0)throw new Error('Multi parallel stage');
  await page.keyboard.down('Alt');
  await node(c.id).dragTo(col('交互系统').locator('[data-drop-gap="1"]'));
  await page.keyboard.up('Alt');
  await waitFor(async()=>{const s=await api('snapshot',{project});return s.tasks.find(t=>t.id===c.id).module==='交互系统';},'Serial drag did not save');
  snapshot=await api('snapshot',{project});
  if(snapshot.tasks.find(t=>t.id===c.id).upstream.length!==2)throw new Error('Next stage not dependent on full parallel group');
  await page.getByRole('button',{name:'切换状态：待归类C',exact:true}).click();
  const menu=page.getByRole('dialog',{name:'快捷流转状态',exact:true});
  if(!await menu.getByRole('button',{name:'进行中',exact:true}).isDisabled())throw new Error('Blocked start enabled');
  await page.keyboard.press('Escape');
  await page.getByRole('button',{name:'查看待归类C',exact:true}).click();
  const dialog=page.getByRole('dialog',{name:'任务详情',exact:true});
  await dialog.waitFor();const box=await dialog.boundingBox();
  if(Math.abs(box.x+box.width/2-720)>3||Math.abs(box.y+box.height/2-500)>3)throw new Error('Task edit not centered: '+JSON.stringify(box));
  await page.getByRole('button',{name:'编辑工作状态',exact:true}).dblclick();
  await page.getByRole('combobox',{name:'工作状态',exact:true}).click();
  if(!await page.getByRole('listbox',{name:'工作状态选项',exact:true}).getByRole('option',{name:'进行中',exact:true}).isDisabled())throw new Error('Inline blocked start enabled');
  await page.keyboard.press('Escape');
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  for(const id of [a.id,b.id]){const t=(await api('snapshot',{project})).tasks.find(t=>t.id===id);await api('edit',{id,revision:t.revision,patch:{state:'done'}});}
  await page.reload();
  await page.getByRole('button',{name:'查看待归类C',exact:true}).focus();await page.keyboard.press('s');
  await page.getByRole('dialog',{name:'快捷流转状态',exact:true}).getByRole('button',{name:'进行中',exact:true}).click();
  await waitFor(async()=>await node(c.id).locator('.flow-node-detail').count()===1,'Quick status did not start');
  let t=(await api('snapshot',{project})).tasks.find(t=>t.id===a.id);await api('edit',{id:a.id,revision:t.revision,patch:{state:'todo'}});
  await page.reload();await node(c.id).getByText('上游未完成 · 保留当前进度',{exact:true}).waitFor();
  await col('交互系统').locator('.workflow-header').click({button:'right'});
  await page.getByRole('menuitem',{name:'编辑名称和优先级',exact:true}).click();
  await page.getByRole('textbox',{name:'需求列名称',exact:true}).fill('新交互系统');
  await choose('优先级','低');
  await page.getByRole('button',{name:'保存需求列',exact:true}).click();await col('新交互系统').waitFor();
  snapshot=await api('snapshot',{project});
  if(snapshot.tasks.filter(t=>[a.id,b.id,c.id].includes(t.id)).some(t=>t.module!=='新交互系统'))throw new Error('Rename did not move members');
  await page.keyboard.down('Alt');
  await node(c.id).dragTo(node(a.id));
  await page.keyboard.up('Alt');
  await waitFor(async()=>{const s=await api('snapshot',{project});return s.tasks.find(t=>t.id===c.id).upstream.length===0;},'Drop on node did not merge parallel');
  await page.reload();await col('新交互系统').waitFor();
  if(await col('新交互系统').locator('.flow-stage').count()!==1)throw new Error('Parallel stage not persisted');
  await page.keyboard.down('Alt');
  if(!await page.locator('.workflow-board.is-arranging').count())throw new Error('Alt mode not entered');
  await page.keyboard.up('Alt');
  if(await page.locator('.workflow-board.is-arranging').count())throw new Error('Alt release not exited');
  await page.screenshot({path:'output/playwright/workflow-arrangement.png'});
  return 'Empty columns, priority/rename, actual multi drag, serial/parallel, blocked start, active preservation, S menu, centered modal and persistence passed';
}
