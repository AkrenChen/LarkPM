async page => {
  await page.setViewportSize({width:1440,height:1000});
  const api=async(method,params={})=>{const r=await page.request.post(new URL('/rpc',page.url()).href,{data:{method,params}});const v=await r.json();if(v.error)throw new Error(v.error);return v.result;};
  const {project}=await api('demo');
  const a=await api('create',{project,title:'拖动取消A',module:'编排验收'}),b=await api('create',{project,title:'拖动取消B',module:'编排验收'});
  for(let i=0;i<22;i++)await api('create',{project,title:'长列节点'+i,module:'长列'});
  await page.evaluate(id=>localStorage.setItem('larkpm-project',id),project);await page.reload();
  const controls=await page.getByRole('textbox',{name:'搜索任务或模块',exact:true}).boundingBox();
  const newColumn=await page.getByRole('button',{name:'新建需求列',exact:true}).boundingBox();
  if(Math.abs(controls.y-newColumn.y)>4)throw new Error('Filter and add column not on same row');
  const board=page.locator('.workflow-board');
  let box=await board.boundingBox();
  if(1000-box.y-box.height<20)throw new Error('No bottom gutter');
  if(!await board.evaluate(el=>el.scrollHeight>el.clientHeight))throw new Error('Tall columns do not scroll inside board');
  await board.evaluate(el=>el.scrollTop=el.scrollHeight);
  box=await board.boundingBox();if(1000-box.y-box.height<20)throw new Error('Bottom gutter lost after scrolling');
  await page.screenshot({path:'output/playwright/workflow-bottom-gutter.png'});
  await board.evaluate(el=>{el.scrollTop=0;el.scrollLeft=0;});
  const node=id=>page.locator('[data-task-id="'+id+'"]');
  await node(a.id).scrollIntoViewIfNeeded();
  const before=JSON.stringify((await api('snapshot',{project})).tasks);
  await page.keyboard.down('Alt');
  const src=await node(a.id).boundingBox(),dst=await node(b.id).boundingBox();
  await page.mouse.move(src.x+40,src.y+15);await page.mouse.down();
  await page.mouse.move(src.x+55,src.y+20,{steps:6});await page.mouse.move(dst.x+50,dst.y+15,{steps:10});
  await page.keyboard.press('Escape');await page.mouse.up();await page.keyboard.up('Alt');
  if(JSON.stringify((await api('snapshot',{project})).tasks)!==before)throw new Error('Esc modified saved arrangement');
  await page.keyboard.down('Alt');
  await page.evaluate(()=>window.dispatchEvent(new Event('blur')));
  if(await page.locator('.workflow-board.is-arranging').count())throw new Error('Blur retained mode');
  await page.keyboard.up('Alt');
  await page.getByRole('button',{name:'查看拖动取消A',exact:true}).click();
  await page.getByRole('textbox',{name:'任务标题',exact:true}).focus();await page.keyboard.down('Alt');
  if(await page.locator('.workflow-board.is-arranging').count())throw new Error('Shortcut active while editing');
  await page.keyboard.up('Alt');await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  await page.setViewportSize({width:1920,height:1080});
  await page.screenshot({path:'output/playwright/workflow-controls-wide.png'});
  return 'Merged controls, fixed bottom gutter, internal long column scrolling, native Escape cancellation, blur cleanup and input shortcut isolation passed';
}
