async page => {
  const chooseDepartment=async(label,name)=>{await page.getByRole('combobox',{name:label,exact:true}).click();await page.getByRole('listbox',{name:label+'选项',exact:true}).getByRole('option',{name,exact:true}).click();};
  const api = async (method,params={}) => {
    const r=await page.request.post(new URL('/rpc',page.url()).href,{data:{method,params}});
    const value=await r.json(); if(value.error) throw new Error(value.error); return value.result;
  };
  const a=await api('project_save',{name:'部门回归 A',base_token:'fixture',table_id:'one'});
  const b=await api('project_save',{name:'部门回归 B',base_token:'fixture',table_id:'other'});
  const task=await api('create',{project:a.id,title:'部门下拉回归',owners:[{id:'a',name:'测试甲'}]});
  const settings=(await api('snapshot')).settings;
  await api('settings_save',{...settings,people:[{id:'a',name:'测试甲',department:'旧部门'},{id:'b',name:'测试乙',department:''}]});
  await page.evaluate(id=>localStorage.setItem('larkpm-project',id),a.id);
  await page.reload();
  await page.getByRole('button',{name:'设置',exact:true}).click();
  await page.getByRole('button',{name:'人员与部门',exact:true}).click();
  const single=page.getByRole('combobox',{name:'测试甲的默认部门（a）',exact:true});
  await single.click();
  const menu=page.getByRole('listbox',{name:'测试甲的默认部门（a）选项',exact:true});
  await menu.getByRole('option',{name:'动效',exact:true}).waitFor();
  if(!(await single.innerText()).startsWith('旧部门')) throw new Error('Legacy department cleared');
  const names=await menu.getByRole('option').allTextContents();
  if(JSON.stringify(names.slice(1))!==JSON.stringify(['策划','美术','UI','开发','动效'])) throw new Error('Source options/order mismatch');
  await menu.getByRole('option',{name:'UI',exact:true}).click();
  if(await single.locator('.department-tag').evaluate(el=>getComputedStyle(el).backgroundColor)!=='rgb(255, 248, 223)') throw new Error('Department color missing');
  await page.getByRole('checkbox',{name:'选择测试甲（a）',exact:true}).check();
  await page.getByRole('checkbox',{name:'选择测试乙（b）',exact:true}).check();
  await chooseDepartment('批量部门','开发');
  await page.getByRole('button',{name:'应用部门',exact:true}).click();
  await page.getByRole('button',{name:'保存设置',exact:true}).click();
  await page.getByText('设置已保存 · 仅保存在本机',{exact:true}).waitFor();
  await page.reload();
  await page.getByRole('button',{name:'设置',exact:true}).click();
  await page.getByRole('button',{name:'人员与部门',exact:true}).click();
  if(!(await single.innerText()).startsWith('开发')) throw new Error('Batch department not persisted');
  await page.getByRole('combobox',{name:'当前项目',exact:true}).selectOption(b.id);
  await page.getByText('选项来源：部门回归 B · 部门',{exact:true}).waitFor();
  await single.click();
  await menu.getByRole('option',{name:'测试部',exact:true}).waitFor();
  if(await menu.getByRole('option',{name:'UI',exact:true}).count()) throw new Error('Another project options leaked');
  if(!(await single.innerText()).startsWith('开发')) throw new Error('Project switch destroyed shared default');
  await single.press('Escape');
  await page.getByRole('combobox',{name:'当前项目',exact:true}).selectOption(a.id);
  await single.click();
  await menu.getByRole('option',{name:'UI',exact:true}).waitFor();
  await single.press('Escape');
  await page.screenshot({path:'output/playwright/departments-settings.png'});
  await page.getByRole('button',{name:'需求总览',exact:true}).click();
  await page.locator('.task-card').filter({hasText:task.title}).click();
  await page.getByRole('checkbox',{name:'使用人员默认部门',exact:true}).uncheck();
  await chooseDepartment('手动指定部门','UI');
  await page.getByRole('button',{name:'保存修改',exact:true}).click();
  await page.getByText('任务已保存于本地',{exact:true}).waitFor();
  const saved=await api('snapshot',{project:a.id});
  if(JSON.stringify(saved.tasks[0].departments)!==JSON.stringify(['开发','UI'])) throw new Error('Multiple departments lost');
  await page.getByRole('button',{name:'关闭任务详情',exact:true}).click();
  await page.locator('.task-card').filter({hasText:task.title}).click();
  await page.getByRole('button',{name:'移除部门开发',exact:true}).click();
  await page.getByRole('button',{name:'保存修改',exact:true}).click();
  await page.getByText('任务已保存于本地',{exact:true}).waitFor();
  if(JSON.stringify((await api('snapshot',{project:a.id})).tasks[0].departments)!==JSON.stringify(['UI'])) throw new Error('Remove department failed');
  await page.screenshot({path:'output/playwright/departments-task.png'});
  return 'source options/order/color, legacy preservation, batch save, project isolation, multi-department and removal passed';
}
